def init_db():
    conn=get_db();c=conn.cursor()
    c.execute("""CREATE TABLE IF NOT EXISTS users(id INTEGER PRIMARY KEY,username TEXT UNIQUE,password TEXT,is_admin INT DEFAULT 0,avatar TEXT,xp INT DEFAULT 0,level INT DEFAULT 1,title TEXT DEFAULT 'Rookie 1',streak_count INT DEFAULT 0,last_checkin_date TEXT,last_pr_update_date TEXT,weight REAL,last_weighin_date TEXT);""")
    c.execute("""CREATE TABLE IF NOT EXISTS prs(user_id INT PRIMARY KEY,squat INT DEFAULT 0,bench INT DEFAULT 0,deadlift INT DEFAULT 0);""")
    c.execute("""CREATE TABLE IF NOT EXISTS activity(id INTEGER PRIMARY KEY,user_id INT,type TEXT,detail TEXT,xp_awarded INT DEFAULT 0,created_at TEXT);""")
    c.execute("""CREATE TABLE IF NOT EXISTS challenge_pool(id INTEGER PRIMARY KEY,text TEXT);""")
    c.execute("""CREATE TABLE IF NOT EXISTS user_daily_challenges(id INTEGER PRIMARY KEY,user_id INT,date TEXT,challenge_id INT,completed INT DEFAULT 0);""")
    c.execute("""CREATE TABLE IF NOT EXISTS progress_photos(id INTEGER PRIMARY KEY,user_id INT,image_path TEXT,created_at TEXT);""")
    c.execute("""CREATE TABLE IF NOT EXISTS crew_state(id INTEGER PRIMARY KEY CHECK(id=1),current_goal INT DEFAULT 5000);""")
    c.execute("INSERT OR IGNORE INTO crew_state(id,current_goal)VALUES(1,5000)");conn.commit();conn.close()

def login_required(f):
    @wraps(f)
    def w(*a,**k):
        if"user_id"not in session:return redirect(url_for("login"))
        return f(*a,**k)
    return w
def admin_required(f):
    @wraps(f)
    def w(*a,**k):
        if"username"not in session or session.get("username")!="admin":return redirect(url_for("login"))
        return f(*a,**k)
    return w

TITLE_BUCKETS=["Rookie 1","Rookie 2","Rookie 3","Amateur","Veteran","Meathead","Beast","Hulk","Olympian","Titan"]
def xp_needed_for_level(l):return int(100*(1.6**(l-1)))if l<50 else 10**9
def recalc_level(x):
    l,r=1,x
    while l<50:
        need=xp_needed_for_level(l)
        if r>=need:r-=need;l+=1
        else:break
    return l
def title_for_level(l):
    if l>=50:return"Immortal"
    return TITLE_BUCKETS[(l-1)//5]
